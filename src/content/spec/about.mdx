import { getCollection } from "astro:content";
import { siteConfig } from "@/config";
import { getSortedPosts } from "@/utils/content-utils";

export const meta = await (async () => {
	const posts = await getSortedPosts();
	const start = siteConfig.siteStartDate || "2025-01-01";
	const days = Math.max(1, Math.floor((Date.now() - new Date(start).getTime()) / 86400000) + 1);
	const latest = posts.reduce((m, p) => {
		const d = p.data.updated && p.data.updated > p.data.published ? p.data.updated : p.data.published;
		return !m || d > m ? d : m;
	}, null);
	const ago = latest ? Math.max(0, Math.floor((Date.now() - latest.getTime()) / 86400000)) : null;
	const shuoshuo = await getCollection("shuoshuo");
	return { count: posts.length, days, ago, shuoshuo: shuoshuo.length };
})();
